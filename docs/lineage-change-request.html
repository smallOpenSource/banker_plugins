def split_agent_message(text):
    # 부분 문자열 검사 금지. 첫머리 구조 일치여야 한다 — 사용자가 태그를
    # '언급만' 해도 에이전트로 뒤바뀌고, 인용한 래퍼 내용이 사용자 문장을 대체한다.
    m = _AGENT_WRAP_RE.match(_PEER_LEAD_RE.sub("", text.lstrip()))
    if not m: return None, None
    who = _AGENT_FROM_RE.search(m.group(2) or "")
    # 하네스 안내문은 닫는 태그 뒤에 붙으므로 내부 텍스트만 취하면 자동 제외된다.
    # 미리 잘라내면 그 문구를 인용한 보고서가 닫는 태그까지 함께 잘린다.
    body = m.group(3).strip()
    if body.startswith('{"type"'): return (who.group(1) if who else "agent"), ""  # idle 알림 → 폐기
    return (who.group(1) if who else "agent"), body

def clean_user_text(text, drop_harness=True):
    out = _NOISE_CLOSE_RE.sub("", _NOISE_BLOCK_RE.sub("", text)).strip()

    mb = re.match(r"^/([\w-]+)\s*$", out)          # 래퍼 없이 맨 명령만 온 레코드
    if mb and drop_harness and mb.group(1) in _HARNESS_CMDS: return ""

    if out:
        # 무언가 살아남았다 = 태그를 인용한 메시지다. 그대로 둔다.
        # 실측: 진짜 명령 레코드는 141바이트짜리 순수 래퍼였다.
        return "" if _NOISE_OPEN_RE.match(out) else out

    m = re.search(r"<command-name>([\s\S]*?)</command-name>", text)
    if not m: return ""
    name = m.group(1).strip()
    if drop_harness and ":" not in name and name.lstrip("/") in _HARNESS_CMDS: return ""
    a = re.search(r"<command-args>([\s\S]*?)</command-args>", text)
    return (name + (" " + a.group(1).strip() if a and a.group(1).strip() else "")).strip()

def drop_echo_exchanges(turns):
    # 아무 일도 하지 않은 교환은 구조로 잡는다: 짧은 질문 + 도구 0건의 토큰 답변.
    # 문턱 5의 근거 — 62MB 안에서 도구 없는 40자 이하 답변은 "1"(1자)과
    # "Login expired…"(33자) 둘뿐이었고, 한국어 확인("네, 맞습니다." 8자)은 살려야 한다.
    # 답을 지우면 질문도 함께 사라진다.
    drop = set()
    for i, t in enumerate(turns):
        if t["role"] != "user" or len(t["text"]) > ECHO_ASK: continue
        n = turns[i+1] if i+1 < len(turns) else None
        if (n and n["role"] == "assistant" and not n["tools"]
                and len(n["text"].strip()) <= ECHO_REPLY
                and n.get("session") == t.get("session")):   # --all-sessions 가드
            drop |= {i, i+1}
    return [t for i, t in enumerate(turns) if i not in drop]

    # 반드시 --hide-tool-only 앞에서 수행.
    # 순수 함수 — 입력을 변형하지 않아야 두 번 호출해도 도구가 중복 집계되지 않는다.

def summarize_turn(turn):
    parts = [p for p in turn.get("parts") or [] if p.strip()]
    if len(parts) < 2:
        if len(turn["text"]) < SINGLE_SPLIT_MIN: return naive_summary(turn["text"])
        parts = [b for b in re.split(r"\n\s*\n", turn["text"]) if b.strip()] or parts
        if len(parts) < 2: return naive_summary(turn["text"])
    head = naive_summary(parts[0])
    # 도입이 순수 의도문이면 결론에 자리를 통째로 내준다
    if _INTENT_ONLY_RE.match(head):
        only = naive_summary(_tail_block(parts), from_end=True)
        if only and only != "(empty turn)": return _cut(only, 120)
    # 결론 절은 마지막 블록을 끝에서부터 읽는다. 앞에서 읽으면 결론이 아니라
    # '두 번째 도입'이 나온다 — 접히는 턴의 46%가 마지막 블록 400자 초과였다.
    tail = naive_summary(_tail_block(parts), from_end=True)
    if not tail or tail == head: return _cut(head, 120)       # 자르기 전 값끼리 비교할 것
    head = _cut(head, 58)
    return head + " … " + _cut(tail, 120 - len(head) - 3)

def _tail_block(blocks):
    # 표·헤딩·코드펜스·15자 미만은 건너뛴다 — 결론 자리에 "| a | b |" 가 오지 않게.

        s = b.strip()
        if len(s) < 15 or s.startswith(("|", "```")) or re.match(r"^#{1,6}\s", s): continue

    return blocks[-1]

불변식 재구현할 때 다시 깨지기 쉬운 것

아래는 구현 중 실제로 두 번 이상 반대 방향으로 잘못 짰던 지점입니다. 리팩터링 시 가장 먼저 되돌아가는 자리라 따로 적습니다.

1 · 위치가 아니라 형태로 판정할 것 — 단, 명령 레코드 자체는 반대
주입 본문을 “명령 다음 레코드”로 잡으면 사용자의 정정·취소가 지워집니다. 반대로 명령 레코드는 래퍼를 걷어낸 잔여가 비었는가로 봐야 합니다 — 첫머리 위치로 판정하면 태그를 인용한 진짜 질문이 /copy 한 줄로 증발합니다.
2 · 사용자 글을 지우는 규칙을 추가할 땐 삭제 대조를 함께 낼 것
무엇이 몇 건 사라졌고 전부 노이즈였는지 표로 확인해야 합니다. “아마 노이즈일 것”으로 지우면 되돌릴 수 없습니다.
3 · 어떤 변환 단계도 앞 단계가 내놓은 마크업 위로 다시 돌지 말 것
정규식을 순차 적용하면 앞서 만든 href·</strong>를 다음 규칙이 다시 건드립니다.
4 · 문단 누적기는 한 줄도 못 담으면 강제로 한 줄을 소비할 것
“블록”으로 판정되는데 어떤 분기도 처리하지 않으면 인덱스가 전진하지 않아 무한 루프가 됩니다.
5 · 요약기를 고치면 캐시 버전을 올릴 것
본문 해시만으로는 알고리즘 변경을 감지하지 못해 한 실행에 두 알고리즘의 요약이 섞입니다.
6 · 검증은 PDF·화면이 아니라 산출물의 소스에서 할 것
태그 균형·중첩은 눈으로 보이지 않고, 브라우저는 깨진 마크업을 조용히 복구해 증상을 감춥니다. self_verify가 렌더러 출력 전체를 덮어야 합니다.
